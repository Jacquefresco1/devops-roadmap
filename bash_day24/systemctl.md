systemctl - управление сервисами systemd
systemd - система, которая управляет сервисами(службами) на большинстве современных Linux. Другими словами, это то, что запускает/останавливает/перезапускает программы, которые работают постоянно в фоновом режиме.
systemctl status  "название программы"           # посмотреть статус сервиса (работает/остановлен/ошибка)
systemctl start   "название программы"            # запустить
systemctl stop    "название программы"             # остановить
systemctl restart "название программы"          # перезапустить (стоп + старт)
systemctl enable  "название программы"           # включить автозапуск при загрузке системы
systemctl disable "название программы"          # выключить автозапус

systemctl list-unit --type=service - покажет список зареестрированых служб(сервисов) и их статус(failed/active/inactive)
#Практика на сron
systemctl status cron - смотрим на сron, убеждаемся что он работает
sudo systemctl stop cron - останавливаем его
systemctl status cron - смотрим, что он остановился
sudo system start cron - запускаем обратно
systemctl list-units --type=service - посмотреть списиок сеервисов
