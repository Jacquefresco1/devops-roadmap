kill "PID" - остановка процесса по ID.
kill -15;9;1 - (-15 - SIGTERM-> стандартная просьба о закрытии процесса, используется в 100% случаев; -9 SIGKILL->немедленное, принудительное завершение процесса, нужно в случае, если после -15, процесс не закрылся; -1 SIGHUB-> процесс перечитывает конфиг, без остановки.
killall "название процесса" - используется, если процессов с одним именем несколько, завершает все древо процессов.
#Практика
sleep 300 & - запускаем процесс ожидания, "&"-> отвечает за запуск в фоновом режиме.
ps aux | grep sleep - убедимся, что процесс запущем и смотрим его PID
kill -15 "PID" - закрываем процесс, с помощью SIGTERM
ps aux | grep sleep - теперь видим только эхо поиска.
