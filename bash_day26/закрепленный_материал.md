## Скрипты с файлами и find
- stat -c "%a,%U,%G" файл — права числом, владелец, группа одной командой; %n — имя файла
- files=*.$2 без кавычек: * раскрывается сразу, в текущей папке. Правильно: files="*.$2", тогда * доходит до find -name
- for file in $find1 — рвёт имена с пробелами ("test file.log" → "test" и "file.log"), проверено вживую
- for file in "$find1" — схлопывает весь список в ОДНУ итерацию (хуже). Для имён с пробелами нужен find -print0 | while IFS= read -r -d '' file (пока не проходили)
- имена файлов с пробелами бывают (скачанные файлы, чужие программы), в Linux это разрешено

## Копия с датой
- date +%Y-%m-%d — дата в формате 2026-10-08
- cp всегда требует ДВА аргумента: что копировать и куда. cp "$(date +%Y-%m-%d)_$1" без источника — ошибка
- touch создаёт пустой файл, это не копия
- дата перед расширением (file_2026-10-08.txt) требует отрезать расширение (${var%.*}) — не проходили
- с путём вида папка/файл дата прицепится к началу всего пути — осторожно

## awk
- максимум: $7 > max { max=$7 } END { print max } — max без объявления стартует с 0; хранит только число, а не строки с ним
- NR>1 надёжнее, чем фильтр по тексту заголовка ($1!="USER"): не зависит от содержимого
- массив без объявления: { count[$1]++ } END { for (k in count) print k, count[k] } — счётчик на каждый ключ, порядок вывода не гарантирован
- awk -F ':' '$NF=="/bin/bash" { print $1 }' /etc/passwd — имена пользователей с bash
- фильтр "начинается с буквы" ловит мусор в тестовом файле, но не надёжен: надёжнее фильтровать по формату IP

## grep и подсчёт
- grep -c '^$' — пустые строки (строго 0 символов; строка из пробелов НЕ пустая; для пробелов: '^ *$')
- grep -vc '^$' — непустые; сумма двух равна общему числу строк
- grep -c "#" ловит # где угодно, grep -c "^#" — только комментарии в начале. Разница видна на "port=80 # note"
- grep -c только считает; вычитание делает скрипт через $(( ))
- в $(( )) переменные без $ и без кавычек: $(( total - comments ))
- # в имени переменной начинает комментарий — count_# не работает

## if и коды возврата
- [ "$a"="b" ] без пробелов вокруг = — ОДНА непустая строка, всегда истинно. Правильно: [ "$a" = "b" ]
- if команда; then — смотрит на КОД ВОЗВРАТА, не на текст вывода. grep: 0 нашёл (хоть одну строку), 1 не нашёл
- if ss -tuln | grep -F ':22 ' &>/dev/null; then — проверка порта без переменных и без awk
- ss -tuln: l = только слушающие, поэтому строка с портом уже значит "слушается"
- ':22 ' с пробелом, иначе цепляет 2200, 22222
- после "Error" в else нужен exit 1, иначе скрипт завершится с кодом 0 ("успех")
